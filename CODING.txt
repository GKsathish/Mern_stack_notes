Q1. Reverse a string
--------------------
Input: "hello" -> Output: "olleh"

1) Brute Force:
function reverseString(str) {
  let rev = "";
  for (let i = 0; i < str.length; i++) {
    rev = str[i] + rev;
  }
  return rev;
}
Time: O(n^2), Space: O(n)

2) Built-in:
const reverseString = str => str.split("").reverse().join("");
Time: O(n), Space: O(n)

3) Two Pointers:
function reverseString(str) {
  const chars = str.split("");
  let left = 0, right = chars.length - 1;
  while (left < right) {
    [chars[left], chars[right]] = [chars[right], chars[left]];
    left++; right--;
  }
  return chars.join("");
}
Time: O(n), Space: O(n)

Q2. Reverse words in a sentence
-------------------------------
Input: "hello world js" -> Output: "js world hello"

1) Brute Force:
function reverseWords(str) {
  let words = [], word = "";
  for (let ch of str) {
    if (ch === " ") { if (word) { words.unshift(word); word = ""; } }
    else word += ch;
  }
  if (word) words.unshift(word);
  return words.join(" ");
}

2) Built-in:
const reverseWords = str => str.trim().split(/\s+/).reverse().join(" ");

3) Optimized:
Split once, then use two pointers to swap words in the array before joining.
Time: O(n), Space: O(n)

Q3. Palindrome check (ignore case, spaces, special chars)
---------------------------------------------------------
Input: "A man, a plan, a canal: Panama" -> true

1) Brute Force:
function isPalindrome(str) {
  let clean = "";
  for (let ch of str.toLowerCase()) {
    if ((ch >= "a" && ch <= "z") || (ch >= "0" && ch <= "9")) clean += ch;
  }
  let rev = "";
  for (let i = clean.length - 1; i >= 0; i--) rev += clean[i];
  return clean === rev;
}

2) Built-in:
const isPalindrome = str => {
  const clean = str.toLowerCase().replace(/[^a-z0-9]/g, "");
  return clean === clean.split("").reverse().join("");
};

3) Two Pointers:
function isPalindrome(str) {
  let left = 0, right = str.length - 1;
  while (left < right) {
    while (left < right && !/[a-z0-9]/i.test(str[left])) left++;
    while (left < right && !/[a-z0-9]/i.test(str[right])) right--;
    if (str[left].toLowerCase() !== str[right].toLowerCase()) return false;
    left++; right--;
  }
  return true;
}
Time: O(n), Space: O(1)

Q4. Anagram check
-----------------

Sort both strings and compare.
function isAnagram(a, b) {
  return a.split("").sort().join("") === b.split("").sort().join("");
}
Time: O(n log n)

2) Built-in + Map:
Use a Map to count characters in first string, then reduce counts for second string.

3) Optimized Frequency Array:
function isAnagram(a, b) {
  if (a.length !== b.length) return false;
  const count = {};
  for (const ch of a) count[ch] = (count[ch] || 0) + 1;
  for (const ch of b) {
    if (!count[ch]) return false;
    count[ch]--;
  }
  return true;
}
Time: O(n), Space: O(k)

Q9. Longest substring without repeating characters
--------------------------------------------------
Input: "abcabcbb" -> 3

Check every substring and verify whether all characters are unique.

3) Sliding Window:
function lengthOfLongestSubstring(str) {
  const seen = new Set();
  let left = 0, max = 0;
  for (let right = 0; right < str.length; right++) {
    while (seen.has(str[right])) {
      seen.delete(str[left]);
      left++;
    }
    seen.add(str[right]);
    max = Math.max(max, right - left + 1);
  }
  return max;
}
Time: O(n), Space: O(k)

Q19. Longest palindromic substring
----------------------------------
1) Brute Force:
Check every substring whether palindrome.
Time: O(n^3)

2) Built-in + Helper:
Use a helper isPalindrome and check substrings from longest to shortest.

3) Expand Around Center:
function longestPalindrome(str) {
  let start = 0, end = 0;
  const expand = (l, r) => {
    while (l >= 0 && r < str.length && str[l] === str[r]) { l--; r++; }
    return r - l - 1;
  };
  for (let i = 0; i < str.length; i++) {
    const len = Math.max(expand(i, i), expand(i, i + 1));
    if (len > end - start) {
      start = i - Math.floor((len - 1) / 2);
      end = i + Math.floor(len / 2);
    }
  }
  return str.slice(start, end + 1);
}
Time: O(n^2), Space: O(1)

Q20. Find all permutations of a string
--------------------------------------
Input: "abc" -> ["abc","acb","bac","bca","cab","cba"]

1) Brute Force:
Generate all possible arrangements by recursion and filter duplicates.

2) Built-in:
Use recursion with remaining characters and current path.

3) Backtracking:
function permutations(str) {
  const result = [];
  const backtrack = (path, remaining) => {
    if (!remaining.length) return result.push(path.join(""));
    for (let i = 0; i < remaining.length; i++) {
      backtrack(
        [...path, remaining[i]],
        [...remaining.slice(0, i), ...remaining.slice(i + 1)]
      );
    }
  };
  backtrack([], [...str]);
  return result;
}
Time: O(n * n!)

Q21. Check if two strings are isomorphic
----------------------------------------
Input: "egg", "add" -> true

1) Brute Force:
For every character pair, verify consistent mapping both ways.

2) Built-in:
Use two Maps for forward and reverse mapping.

3) Optimized:
function isIsomorphic(s, t) {
  if (s.length !== t.length) return false;
  const mapST = {}, mapTS = {};
  for (let i = 0; i < s.length; i++) {
    const a = s[i], b = t[i];
    if (mapST[a] && mapST[a] !== b) return false;
    if (mapTS[b] && mapTS[b] !== a) return false;
    mapST[a] = b;
    mapTS[b] = a;
  }
  return true;
}
Time: O(n)

function binarySearch(arr, target) {
  let left = 0, right = arr.length - 1;
  while (left <= right) {
    const mid = Math.floor((left + right) / 2);
    if (arr[mid] === target) return mid;
    if (arr[mid] < target) left = mid + 1;
    else right = mid - 1;
  }
  return -1;
}

Q25. Implement map, filter, reduce polyfills
--------------------------------------------
Array.prototype.myMap = function(cb) {
  const result = [];
  for (let i = 0; i < this.length; i++) {
    if (i in this) result[i] = cb(this[i], i, this);
  }
  return result;
};

Array.prototype.myFilter = function(cb) {
  const result = [];
  for (let i = 0; i < this.length; i++) {
    if (i in this && cb(this[i], i, this)) result.push(this[i]);
  }
  return result;
};

Array.prototype.myReduce = function(cb, initialValue) {
  let acc = initialValue;
  let start = 0;
  if (acc === undefined) {
    acc = this[0];
    start = 1;
  }
  for (let i = start; i < this.length; i++) {
    if (i in this) acc = cb(acc, this[i], i, this);
  }
  return acc;
};

Q1. Debounce
------------
Run function only after user stops calling it for delay ms.

1) Brute Force:
Use timestamp checks.

2) Built-in:
Use setTimeout and clearTimeout.

3) Standard:
function debounce(fn, delay) {
  let timer;
  return function (...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}

Q2. Throttle
------------
Allow function execution at most once per interval.

1) Brute Force:
Track last execution time.

2) Built-in:
Use setTimeout flag.

3) Standard:
function throttle(fn, limit) {
  let lastCall = 0;
  return function (...args) {
    const now = Date.now();
    if (now - lastCall >= limit) {
      lastCall = now;
      fn.apply(this, args);
    }
  };
}

};

Q7. Promise.all polyfill
------------------------

function promiseAll(promises) {
  return new Promise((resolve, reject) => {
    const results = [];
    let completed = 0;
    promises.forEach((p, i) => {
      Promise.resolve(p).then(value => {
        results[i] = value;
        completed++;
        if (completed === promises.length) resolve(results);
      }, reject);
    });
  });
}

5. JAVASCRIPT: OUTPUT QUESTIONS
=====================================================

Q1. var vs let in loop with setTimeout
--------------------------------------
var creates function-scoped variable, so all callbacks see final value.
let creates block-scoped variable per iteration.

Q2. Event loop order
--------------------
Synchronous code -> process.nextTick -> microtasks (Promises) -> macrotasks (setTimeout).

Q3. this in arrow vs normal function
------------------------------------
Arrow functions inherit this from enclosing scope.
Normal functions get this based on how they are called.

Q4. Closure counter
-------------------
Inner function retains access to outer function variable even after outer function returns.

Q5. Hoisting and TDZ
--------------------
var declarations are hoisted with undefined.
let/const are hoisted but remain in Temporal Dead Zone until initialization.

Q6. == vs ===, [] + {}, typeof null
------------------------------------
== performs type coercion; === does not.
typeof null === "object" is a historical JavaScript bug.

Q7. Prototype chain
-------------------
Objects look up properties on themselves, then their prototype, and continue upward.

Q8. async/await execution order
-------------------------------
await pauses only inside the async function; remaining code becomes microtask.

Q9. Object reference vs copy
----------------------------
Objects are copied by reference; spread creates shallow copy only.

Q10. new keyword and constructor functions
------------------------------------------
new creates an object, sets prototype, binds this, and returns the object.
